          class MainActivity : AppCompatActivity() {
              override fun onCreate(s: Bundle?) {
                  super.onCreate(s)
                  val root = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL; setPadding(48, 96, 48, 48) }
                  root.addView(TextView(this).apply { text = "Какой параметр улучшать?"; textSize = 20f })

                      "Здоровье копейщиков", "Смертоносность стрелков", "Здоровье стрелков")
                  val group = RadioGroup(this)
                  names.forEachIndexed { i, n ->

                  }
                  group.check(ClickService.row)
                  group.setOnCheckedChangeListener { _, id -> ClickService.row = id }
                  root.addView(group)

                  val delayEdit = EditText(this).apply { setText(ClickService.delayMs.toString()) }
                  root.addView(delayEdit)

                  root.addView(Button(this).apply {
                      text = "1. Включить службу доступности"
                      setOnClickListener { startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS)) }

                  root.addView(Button(this).apply {
                      text = "2. Сохранить и показать кнопку"

                          ClickService.maxTries = maxEdit.text.toString().toIntOrNull() ?: 300
                          ClickService.delayMs = delayEdit.text.toString().toLongOrNull() ?: 1300L

                      }
                  })
                  setContentView(root)
              }
          }

          package com.example.wosclicker

          import android.accessibilityservice.AccessibilityService
          import android.accessibilityservice.GestureDescription
          import android.graphics.Bitmap
          import android.graphics.Color
          import android.graphics.Path
          import android.graphics.PixelFormat
          import android.os.Handler
          import android.os.Looper
          import android.view.Display
          import android.view.Gravity
          import android.view.WindowManager
          import android.view.accessibility.AccessibilityEvent
          import android.widget.Button

          class ClickService : AccessibilityService() {

              companion object {
                  var instance: ClickService? = null
                  var row = 4              // 0..5, по умолчанию «Смертоносность стрелков»
                  var maxTries = 300
                  var delayMs = 1300L
                  // Опорные координаты со скриншота 1189x2576
                  private const val REF_W = 1189f
                  private const val REF_H = 2576f
                  private val ROW_Y = floatArrayOf(1228f, 1360f, 1488f, 1618f, 1748f, 1878f)
                  private const val ARROW_X = 1057f
                  private val BTN_RETRY = floatArrayOf(352f, 2272f)
                  private val BTN_REPLACE = floatArrayOf(836f, 2272f)
              }

              private val handler = Handler(Looper.getMainLooper())
              private var running = false
              private var tries = 0
              private var btn: Button? = null

              override fun onServiceConnected() { instance = this; showOverlay() }
              override fun onAccessibilityEvent(e: AccessibilityEvent?) {}
              override fun onInterrupt() { running = false }
              override fun onDestroy() { instance = null; running = false; super.onDestroy() }

              fun showOverlay() {
                  if (btn != null) return
                  val wm = getSystemService(WINDOW_SERVICE) as WindowManager
                  val b = Button(this).apply {
                      text = "START"
                      setOnClickListener {
                          running = !running
                          if (running) { tries = 0; step() } else text = "START"
                      }
                  }
                  val lp = WindowManager.LayoutParams(
                      WindowManager.LayoutParams.WRAP_CONTENT, WindowManager.LayoutParams.WRAP_CONTENT,
                      WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY,
                      WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE, PixelFormat.TRANSLUCENT
                  ).apply { gravity = Gravity.TOP or Gravity.START; x = 20; y = 250 }
                  wm.addView(b, lp)
                  btn = b
              }

              private fun stop(msg: String) { running = false; btn?.text = msg }

              private fun step() {
                  if (!running) return
                  if (tries >= maxTries) { stop("DONE"); return }
                  btn?.text = "STOP ($tries)"
                  takeScreenshot(Display.DEFAULT_DISPLAY, mainExecutor, object : TakeScreenshotCallback {
                      override fun onSuccess(r: ScreenshotResult) {
                          val hw = r.hardwareBuffer
                          val bmp = Bitmap.wrapHardwareBuffer(hw, r.colorSpace)!!.copy(Bitmap.Config.ARGB_8888, false)
                          hw.close()
                          decide(bmp)
                      }
                      override fun onFailure(code: Int) { handler.postDelayed({ step() }, 1500) }
                  })
              }

              private fun decide(bmp: Bitmap) {
                  val w = bmp.width; val h = bmp.height
                  val cx = (ARROW_X / REF_W * w).toInt()
                  val cy = (ROW_Y[row] / REF_H * h).toInt()
                  val half = (0.03f * w).toInt()
                  var green = 0; var red = 0
                  for (x in (cx - half).coerceAtLeast(0) until (cx + half).coerceAtMost(w))
                      for (y in (cy - half).coerceAtLeast(0) until (cy + half).coerceAtMost(h)) {
                          val p = bmp.getPixel(x, y)
                          val r = Color.red(p); val g = Color.green(p); val b = Color.blue(p)
                          if (g > 150 && r < 130 && b < 130) green++
                          else if (r > 170 && g < 110 && b < 110) red++
                      }
                  bmp.recycle()

                  val target = if (green > 40 && green > red) BTN_REPLACE else BTN_RETRY
                  tap(target[0] / REF_W * w, target[1] / REF_H * h)
                  tries++
                  handler.postDelayed({ step() }, delayMs)
              }

              private fun tap(x: Float, y: Float) {
                  val path = Path().apply { moveTo(x, y) }
                  val g = GestureDescription.Builder()
                      .addStroke(GestureDescription.StrokeDescription(path, 0, 60)).build()
                  dispatchGesture(g, null, null)
              }
          }
          EOF_8
